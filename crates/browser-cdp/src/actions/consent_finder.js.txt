(() => {
  "use strict";
  const norm = (s) => String(s || "").replace(/\s+/g, " ").trim().toLowerCase();

  // Well-known consent management platforms. `box` finds the dialog, `reject` / `accept` are
  // their own button ids / classes. Open shadow roots are searched too.
  const CMPS = [
    { name: "OneTrust", box: "#onetrust-banner-sdk, #onetrust-consent-sdk, #onetrust-pc-sdk",
      reject: "#onetrust-reject-all-handler, .ot-pc-refuse-all-handler, .ot-bnr-refuse-all-handler",
      accept: "#onetrust-accept-btn-handler" },
    { name: "Cookiebot", box: "#CybotCookiebotDialog",
      reject: "#CybotCookiebotDialogBodyButtonDecline, #CybotCookiebotDialogBodyButtonDeclineAll",
      accept: "#CybotCookiebotDialogBodyLevelButtonLevelOptinAllowAll, #CybotCookiebotDialogBodyButtonAccept" },
    { name: "Didomi", box: "#didomi-host, #didomi-popup, .didomi-popup-container, .didomi-notice-popup",
      reject: "#didomi-notice-disagree-button, .didomi-continue-without-agreeing",
      accept: "#didomi-notice-agree-button" },
    { name: "Quantcast", box: ".qc-cmp2-container, #qc-cmp2-container, .qc-cmp2-ui",
      reject: "", accept: ".qc-cmp2-summary-buttons button[mode='primary']" },
    { name: "TrustArc", box: "#truste-consent-track, #consent_blackbar, .truste_box_overlay",
      reject: "#truste-consent-required", accept: "#truste-consent-button" },
    { name: "Usercentrics", box: "#usercentrics-root, #usercentrics-cmp",
      reject: "[data-testid='uc-deny-all-button']", accept: "[data-testid='uc-accept-all-button']" },
    { name: "Klaro", box: ".klaro, .cookie-modal, .cookie-notice",
      reject: ".cn-decline, .cm-btn-decline", accept: ".cn-accept-all, .cm-btn-success, .cm-btn-accept-all" },
  ];

  // A banner that is not a known platform must talk about cookies itself: "privacy", "consent"
  // or "tracking" alone also appear in payment and permission dialogs.
  const COOKIE_WORDS = /cookie|çerez|cerez|gdpr|kvkk|rgpd/;
  const REJECT = [
    "reject", "decline", "deny", "refuse", "disagree", "opt out", "opt-out",
    "only necessary", "necessary only", "only essential", "essential only", "only required",
    "required only", "necessary cookies", "essential cookies", "continue without accepting",
    "continue without agreeing", "without accepting",
    "reddet", "yalnızca gerekli", "yalnızca zorunlu", "sadece gerekli", "sadece zorunlu",
    "yalnızca gerekli olan", "kabul etme", "kabul etmiyorum", "istemiyorum",
    "ablehnen", "nur notwendige", "nur erforderliche", "nur essenzielle", "nur essentielle",
    "refuser", "continuer sans accepter", "uniquement nécessaires", "seulement nécessaires",
    "rechazar", "solo necesarias", "solo las necesarias", "solo esenciales", "denegar",
    "rejeitar", "recusar", "apenas necessários", "apenas essenciais", "somente necessários",
    "rifiuta", "solo necessari",
  ];
  // Only "accept everything" style answers to a cookie question. A plain "yes", "ok", "allow"
  // or "agree" says nothing about what is being agreed to.
  const ACCEPT_ALL = [
    "accept all", "allow all", "agree to all", "accept cookies", "allow cookies",
    "accept and close", "accept & close", "accept and continue", "accept & continue",
    "tümünü kabul", "tüm çerezleri kabul", "hepsini kabul", "çerezleri kabul",
    "alle akzeptieren", "alle zulassen", "tout accepter", "accepter tout",
    "aceptar todas", "aceptar todo", "aceitar tudo", "aceitar todos", "accetta tutti",
  ];
  const ACCEPT_EXACT = [
    "accept", "i accept", "accepter", "akzeptieren", "aceptar", "aceitar", "accetta",
    "kabul et", "kabul ediyorum",
  ];
  // Words that make an "accept"-looking label mean the opposite.
  const ACCEPT_WORDS = ["accept", "agree", "allow", "kabul", "akzeptieren", "zustimmen", "accepter",
    "aceptar", "aceitar", "accetta", "einverstanden", "erlauben", "autoriser", "permitir"];
  const NEGATION = /\b(not|never|no|nicht|kein|keine|nie|jamais)\b|n't|n’t|\bne pas\b|değil|hayır|asla|\bno\b/;
  const NOT_A_CHOICE = [
    "selected", "selection", "seçili", "seçilen", "auswahl", "manage", "settings", "customi",
    "preferences", "ayarlar", "tercih", "einstellung", "paramètres", "configurar", "more options",
    "learn more", "read more", "details", "policy", "politika", "daha fazla", "mehr",
    "save", "kaydet", "speichern", "enregistrer", "guardar",
  ];

  const classify = (label) => {
    if (!label || label.length > 70) return null;
    if (NOT_A_CHOICE.some((w) => label.includes(w))) return null;
    // "Do not accept", "don't allow": the opposite of what the accept word says.
    if (ACCEPT_WORDS.some((w) => label.includes(w)) && NEGATION.test(label)) return "reject";
    if (ACCEPT_ALL.some((w) => label.includes(w))) return "accept";
    if (REJECT.some((w) => label.includes(w))) return "reject";
    if (ACCEPT_EXACT.includes(label)) return "accept";
    return null;
  };

  // `node` and every open shadow root below it (or on it).
  const roots = (node) => {
    const out = [node];
    if (node.nodeType === 1 && node.shadowRoot) out.push(...roots(node.shadowRoot));
    for (const el of node.querySelectorAll("*")) {
      if (el.shadowRoot) out.push(...roots(el.shadowRoot));
    }
    return out;
  };
  const queryAll = (scope, sel) => {
    if (!sel) return [];
    const found = [];
    if (scope.nodeType === 1) {
      try { if (scope.matches(sel)) found.push(scope); } catch { /* bad selector: no match */ }
    }
    for (const s of roots(scope)) {
      try { found.push(...s.querySelectorAll(sel)); } catch { /* bad selector: no match */ }
    }
    return found;
  };

  const visible = (el) => {
    const r = el.getBoundingClientRect();
    if (r.width < 2 || r.height < 2) return false;
    const cs = getComputedStyle(el);
    return cs.visibility !== "hidden" && cs.display !== "none" && Number(cs.opacity || 1) > 0.05;
  };

  // Where a click would take the page. Only "stay on this very page" is acceptable: an empty
  // or in-page (#...) address, a no-op script address, or the page's own address.
  const staysHere = (raw) => {
    const href = String(raw || "").trim();
    if (href === "" || href.startsWith("#")) return true;
    if (/^javascript:\s*(void\(\s*0\s*\)|undefined|;)?\s*;?\s*$/i.test(href)) return true;
    if (/^javascript:/i.test(href)) return false;
    try {
      const u = new URL(href, location.href);
      return u.origin === location.origin && u.pathname === location.pathname &&
        u.search === location.search;
    } catch {
      return false;
    }
  };

  // Never click something that would leave the page: a link (any role, also an ancestor link),
  // a new-tab target, a download, a form button posting somewhere else.
  const safeToClick = (el) => {
    const link = el.closest("a");
    if (link) {
      if (link.hasAttribute("href") && !staysHere(link.getAttribute("href"))) return false;
      const t = (link.getAttribute("target") || "").toLowerCase();
      if (t && t !== "_self") return false;
      if (link.hasAttribute("download")) return false;
    }
    const tag = el.tagName.toLowerCase();
    if (tag === "button" || tag === "input") {
      // The PROPERTIES resolve `formaction` / `formtarget` and fall back to the form's own.
      if (el.formAction && !staysHere(el.formAction)) return false;
      const t = String(el.formTarget || (el.form && el.form.target) || "").toLowerCase();
      if (t && t !== "_self") return false;
    }
    return true;
  };

  const labelOf = (el) =>
    norm(el.innerText || el.value || el.getAttribute("aria-label") || el.getAttribute("title") || "");

  const deepHit = (x, y) => {
    let h = document.elementFromPoint(x, y);
    while (h && h.shadowRoot) {
      const n = h.shadowRoot.elementFromPoint(x, y);
      if (!n || n === h) break;
      h = n;
    }
    return h;
  };

  // Scroll the element into view and return its centre when it is really what a click there
  // would hit.
  const target = (el) => {
    try { el.scrollIntoView({ block: "center", inline: "center", behavior: "instant" }); } catch { /* detached */ }
    const r = el.getBoundingClientRect();
    const x = r.left + r.width / 2;
    const y = r.top + r.height / 2;
    if (x < 0 || y < 0 || x > innerWidth || y > innerHeight) return null;
    const hit = deepHit(x, y);
    if (!hit || !(hit === el || el.contains(hit))) return null;
    return { x, y };
  };

  // A dialog that holds a form, a payment, a login or a permission request is not a cookie
  // question, whatever else it says.
  const HOSTILE_FIELDS = "input[type='password'], [autocomplete^='cc-'], input[name*='card' i], input[name*='iban' i], input[name*='cvv' i], input[name*='cvc' i], iframe[src*='pay' i], iframe[src*='stripe' i], iframe[src*='checkout' i]";
  const TEXT_FIELDS = "form input:not([type]), form input[type='text'], form input[type='email'], form input[type='tel'], form input[type='number'], form input[type='search'], form textarea";
  const HOSTILE_TEXT = /wants to access|would like to access|access your|sign in with|log in with|grant access|authori[sz]e|erişim|giriş yap|payment|ödeme|pay now|checkout|card number|credit card|kart numaras|kredi kartı|bezahlen|paiement|\b\d[\d.,]*\s?(usd|eur|try|gbp|tl)\b|[$€£₺]\s?\d/;
  const hostile = (box) =>
    queryAll(box, HOSTILE_FIELDS).length > 0 || queryAll(box, TEXT_FIELDS).length > 0 ||
    HOSTILE_TEXT.test(norm(box.innerText));

  const BUTTONS = "button, [role='button'], a, input[type='button'], input[type='submit']";
  // Every usable, reachable button of a dialog with the kind of answer it gives.
  const candidatesIn = (box, own) => {
    const found = new Map();
    const add = (el, kind, label) => {
      if (found.has(el) || !visible(el) || !safeToClick(el)) return;
      const pt = target(el);
      if (pt) found.set(el, { el, kind, label, pt });
    };
    if (own) {
      for (const el of queryAll(box, own.reject)) add(el, "reject", labelOf(el) || "reject");
      for (const el of queryAll(box, own.accept)) add(el, "accept", labelOf(el) || "accept");
    }
    for (const el of queryAll(box, BUTTONS)) {
      const label = labelOf(el);
      const kind = classify(label);
      if (kind) add(el, kind, label);
    }
    return Array.from(found.values());
  };

  const reply = (cmp, hit) => ({
    found: true, kind: hit.kind, cmp, label: hit.label.slice(0, 60), x: hit.pt.x, y: hit.pt.y,
    origin: location.origin,
  });
  // Reject over accept; the first one in page order.
  const choose = (cands) => cands.find((c) => c.kind === "reject") || cands.find((c) => c.kind === "accept");
  const unsure = (groups) => ({
    found: false, ambiguous: true, origin: location.origin,
    options: groups.flatMap((g) => g.cands.map((c) => c.label.slice(0, 60))).slice(0, 8),
  });

  let seenBox = false;
  let refused = false;

  // 1. Known platforms: their own ids first, then the button texts inside their dialog.
  const cmpGroups = [];
  for (const cmp of CMPS) {
    const boxes = queryAll(document, cmp.box);
    if (!boxes.length) continue;
    const cands = [];
    for (const box of boxes) {
      if (hostile(box)) { refused = true; continue; }
      for (const c of candidatesIn(box, cmp)) if (!cands.some((o) => o.el === c.el)) cands.push(c);
    }
    if (cands.length) cmpGroups.push({ cmp: cmp.name, cands });
  }
  if (cmpGroups.length > 1) return unsure(cmpGroups);
  if (cmpGroups.length === 1) {
    const [g] = cmpGroups;
    return reply(g.cmp, choose(g.cands));
  }

  // 2. Any banner-like element that talks about cookies.
  const NAMED = "[role='dialog'], [aria-modal='true'], dialog, [id*='cookie' i], [class*='cookie' i], [id*='consent' i], [class*='consent' i], [id*='gdpr' i], [class*='gdpr' i], [id*='cmp' i], [class*='cmp-' i], [id*='privacy' i], [class*='privacy' i], [id*='cerez' i], [class*='cerez' i], [id*='kvkk' i], [class*='kvkk' i]";
  const floating = Array.from(document.querySelectorAll("div, section, aside, footer, header, form"))
    .filter((el) => {
      const p = getComputedStyle(el).position;
      return p === "fixed" || p === "sticky";
    });
  const isDialog = (el) => el.matches("[role='dialog'], [aria-modal='true'], dialog");
  const boxes = Array.from(new Set([...queryAll(document, NAMED), ...floating]))
    .filter(visible)
    .filter((el) => {
      const text = norm(el.innerText);
      if (text.length > 2500 || !COOKIE_WORDS.test(text)) return false;
      const p = getComputedStyle(el).position;
      return p === "fixed" || p === "sticky" || p === "absolute" || isDialog(el);
    });
  let groups = [];
  for (const box of boxes) {
    if (hostile(box)) { refused = true; continue; }
    const cands = candidatesIn(box, null);
    if (cands.length) groups.push({ box, cands });
  }
  // The innermost dialog that holds the buttons speaks for its wrappers.
  groups = groups.filter((g) => !groups.some((o) => o !== g && g.box.contains(o.box)));
  if (groups.length > 1) return unsure(groups);
  if (groups.length === 1) return reply("generic", choose(groups[0].cands));
  if (boxes.length) seenBox = true;

  const frameHint = Array.from(document.querySelectorAll("iframe")).some((f) =>
    /consent|cookie|cmp|privacy|sourcepoint|sp_message|didomi|onetrust|cookiebot|trustarc|usercentrics/i
      .test((f.getAttribute("src") || "") + " " + (f.id || "") + " " + (f.title || "")));
  return { found: false, banner: seenBox, refused, frame_hint: frameHint, origin: location.origin };
})()
